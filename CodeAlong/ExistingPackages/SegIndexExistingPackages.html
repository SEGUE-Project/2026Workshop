Packages

There are currently (at least) 3 packages available to compute the indices of segregation we will use in this session: dissimilarity and entropy.

OasisR

OasisR was deprecated last year but seems to function well again. It provides performs most categorical & ordinal measures of segregation, including two-group, one-group, multigroup, and spatially adjusted indices. In addition, it provides resampling methods to test significance. Its core segregation functions generally use a matrix-oriented structure: rows represent spatial units and columns represent population groups.

library(OasisR)

divseg

divseg performs diversity measures and indices proposed by Massey & Denton along the 5 dimensions of segregation (Massey & Denton, 1988). It uses a data-frame and tidyverse-oriented structure. Group variables are selected from columns, and the package can work directly with sf objects.

library(divseg)

segregation

segregation implements D, exposure and isolation measures, and entropy-based M and H indices. It uses a long-data structure in which the population group, spatial unit, and optional frequency or weight are represented explicitly as variables. The package also includes inference, decomposition, and visualisation tools.

library(segregation)

Dissimilarity

The Index of Dissimilarity measures the unevenness with which two population groups are distributed across a set of spatial units. For two groups x and y distributed across n units:

D = \frac{1}{2}\sum_{i=1}^{n} \left| \frac{x_i}{X} - \frac{y_i}{Y} \right|

For the first comparison we distinguish between people aged 0–25 and everyone aged 26 and older.

OasisR

OasisR contains several versions of the dissimilarity index, including the conventional two-group index.

DIDuncan
function (x) 
{
    x <- segdataclean(as.matrix(x))$x
    result <- matrix(data = 0, nrow = ncol(x), ncol = ncol(x))
    varTotal <- colSums(x)
    for (k1 in 1:(ncol(x) - 1)) for (k2 in (k1 + 1):ncol(x)) result[k1, 
        k2] <- 0.5 * sum(abs(x[, k1]/varTotal[k1] - x[, k2]/varTotal[k2]))
    result <- result + t(result)
    return(round(result, 4))
}

DIDuncan() takes as input a matrix x in which rows represent spatial units and columns represent population groups. The values should be population counts, rather than population shares. When more than two groups are provided, the function returns the dissimilarity index for every pair of groups. ds_dissim
function (.data, .cols, .name, .comp = FALSE) 
{
    .cols <- rlang::enquo(.cols)
    if (missing(.name)) {
        .name <- "v_index"
        ret_t <- FALSE
    }
    else {
        ret_t <- TRUE
    }
    sub <- dplyr::select(drop_sf(.data), !!.cols)
    if (ncol(sub) <= 1) {
        stop(".cols refers to a single column")
    }
    sub <- dplyr::ungroup(dplyr::mutate(dplyr::rowwise(sub), 
        .total = sum(dplyr::c_across(everything()))))
    .T <- sum(sub$.total)
    .P <- sum(sub[[1]])/.T
    out <- dplyr::pull(dplyr::mutate(rowwise_if(sub, .comp), 
        `:=`(!!.name, 0.5 * sum(.data$.total * abs(pick_n(1)/.data$.total - 
            .P))/(.T * .P * (1 - .P)))), !!.name)
    if (ret_t) {
        relocate_sf(dplyr::mutate(.data, `:=`(!!.name, out)))
    }
    else {
        out
    }
}

dissimilarity
function (data, group, unit, weight = NULL, se = FALSE, CI = 0.95, 
    n_bootstrap = 100) 
{
    stopifnot(CI > 0 & CI < 1)
    if (length(unique(data[[group]])) != 2) {
        stop("The D index only allows two distinct groups")
    }
    d <- prepare_data(data, group, unit, weight)
    ret <- dissimilarity_compute(d, group, unit)
    if (se == TRUE) {
        vars <- attr(d, "vars")
        n_total <- sum(d[["freq"]])
        if (all.equal(n_total, round(n_total)) == TRUE) {
            message(paste0(n_bootstrap, " bootstrap iterations on ", 
                n_total, " observations"))
        }
        else {
            stop(paste0("bootstrap with a total sample size that is not an integer is not allowed, ", 
                "maybe scale your weights?"))
        }
        draws <- stats::rmultinom(n_bootstrap, n_total, d[["freq"]]/n_total)
        boot_ret <- lapply(seq_len(n_bootstrap), function(i) {
            if (i%%5 == 0) 
                update_log(bs_n = i, bs_max = n_bootstrap)
            d[, `:=`(freq, as.double(draws[, i]))]
            dissimilarity_compute(d[freq > 0], group, unit)
        })
        close_log()
        boot_ret <- rbindlist(boot_ret)
        ret <- bootstrap_summary(ret, boot_ret, "stat", CI)
        setattr(ret, "bootstrap", boot_ret)
    }
    ret
}

OasisR also provides DMulti(), which implements a genuine multigroup generalisation of D:

DMulti
function (x) 
{
    x <- as.matrix(x)
    x <- segdataclean(x)$x
    result <- 0
    pTotal <- colSums(x)/sum(x)
    II <- sum(pTotal * (1 - pTotal))
    t <- rowSums(x)
    Total <- sum(x)
    p <- x/t
    for (k in 1:ncol(x)) result <- result + t %*% abs(p[, k] - 
        pTotal[k])
    result <- result/(2 * Total * II)
    return(round(result, 4))
}

Adding spatial information

Conventional D depends on the population counts in the spatial units, but not on the locations of those units relative to one another. Rearranging the same grid cells spatially would therefore leave D unchanged.

Morrill’s spatial version of D adjusts the conventional index using contiguity between spatial units. OasisR::DIMorrill() accepts the group matrix together with a binary contiguity matrix.

We create a rook-contiguity matrix for the same grid cells used in the two-group example. spdep is used here only to construct the spatial relationship between cells.

neighbours <- spdep::poly2nb(
  age_two_count_sf,
  queen = FALSE
)

contiguity <- spdep::nb2mat(
  neighbours,
  style = "B",
  zero.policy = TRUE
)
DIMorrill
function (x, c = NULL, queen = FALSE, spatobj = NULL, folder = NULL, 
    shape = NULL) 
{
    x <- as.matrix(x)
    if (is.null(c)) 
        c <- contig(spatobj = spatobj, folder = folder, shape = shape, 
            queen = queen)
    cldata <- segdataclean(x, c = c)
    x <- cldata$x
    c <- cldata$c
    result <- matrix(data = 0, nrow = ncol(x), ncol = ncol(x))
    DI <- DIDuncan(x)
    pij <- array(0, dim = c(ncol(x), nrow(x), nrow(x)))
    for (k1 in 1:(ncol(x) - 1)) for (k2 in (k1 + 1):ncol(x)) {
        for (i in 1:nrow(x)) pij[k1, , i] <- x[i, k1]/(x[i, k1] + 
            x[i, k2])
        for (i in 1:nrow(x)) pij[k1, i, ] <- abs(pij[k1, i, ] - 
            x[i, k1]/(x[i, k1] + x[i, k2]))
        pij[k1, , ][is.nan(pij[k1, , ])] <- 0
        matprovi <- c %*% pij[k1, , ]
        matprovi <- matprovi * diag(nrow(x))
        result[k1, k2] <- DI[k1, k2] - sum(matprovi)/sum(c)
        result[k2, k1] <- result[k1, k2]
    }
    return(round(result, 4))
}

Entropy

We will use the same three packages to compute entropy indices of segregation.

OasisR

HTheil
function (x) 
{
    x <- segdataclean(as.matrix(x))$x
    result <- vector(length = ncol(x))
    E <- matrix(data = 0, nrow = nrow(x), ncol = ncol(x))
    pTotal <- colSums(x)/sum(x)
    px <- x/rowSums(x)
    Etot <- rep(0, ncol(x))
    for (k in 1:ncol(x)) Etot[k] <- pTotal[k] * log(1/pTotal[k]) + 
        (1 - pTotal[k]) * log(1/(1 - pTotal[k]))
    for (k in 1:ncol(x)) {
        E[, k] <- px[, k] * log(1/px[, k]) + (1 - px[, k]) * 
            log(1/(1 - px[, k]))
        if (any(is.nan(E[, k]))) 
            E[, k][is.nan(E[, k])] <- 0
        result[k] <- sum((rowSums(x) * (Etot[k] - E[, k]))/(Etot[k] * 
            sum(x)))
    }
    return(round(result, 4))
}

ds_entropy
function (.data, .cols, .name, .comp = FALSE) 
{
    .cols <- rlang::enquo(.cols)
    if (missing(.name)) {
        .name <- "v_index"
        ret_t <- FALSE
    }
    else {
        ret_t <- TRUE
    }
    sub <- dplyr::select(drop_sf(.data), !!.cols)

        .total = sum(dplyr::c_across(everything()))))
    .T <- sum(sub$.total)
    .P <- sum(sub[[1]])/.T
    .E <- .P * plog(1/.P) + (1 - .P) * plog(1/(1 - .P))

    }
    else {

    }

For a genuine overall multigroup H, we can compare OasisR::HMulti() with the H returned by segregation::mutual_total():

HMulti
function (x) 
{
    x <- as.matrix(x)
    x <- segdataclean(x)$x
    E <- vector("list", ncol(x))
    Total <- sum(x)
    pTotal <- colSums(x)/Total
    tx <- rowSums(x)
    px <- x/tx
    Etot <- sum(pTotal * log(1/pTotal))
    result <- 0
    for (k in 1:ncol(x)) {
        E[[k]] <- tx * px[, k] * log(px[, k]/pTotal[k])
        E[[k]][is.na(E[[k]])] <- 0
        result <- result + sum(E[[k]])
    }
    result <- result/(Etot * Total)
    return(round(result, 4))
}
